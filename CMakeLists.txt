cmake_minimum_required(VERSION 3.24)
project(ikview VERSION 0.7.2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum macOS version for the bundled Homebrew dependencies")
  list(PREPEND CMAKE_PREFIX_PATH "/opt/homebrew/opt/qtdeclarative" "/opt/homebrew/opt/qtbase" "/opt/homebrew/opt/qtsvg" "/opt/homebrew/opt/qtimageformats" "/opt/homebrew" "/opt/homebrew/opt/libarchive" "/usr/local" "/usr/local/opt/libarchive")
  set(QT_ADDITIONAL_PACKAGES_PREFIX_PATH "${CMAKE_PREFIX_PATH}")
endif()
find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets Concurrent Quick PrintSupport)
find_package(LibArchive REQUIRED)
find_package(exiv2 CONFIG QUIET)

add_library(ikview_core STATIC src/document.cpp src/document.h src/nativeimage.cpp src/nativeimage.h)
target_include_directories(ikview_core PUBLIC src)
target_link_libraries(ikview_core PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent LibArchive::LibArchive)
if(APPLE)
  target_link_libraries(ikview_core PRIVATE "-framework ImageIO" "-framework CoreGraphics" "-framework CoreFoundation" "-framework CoreServices")
endif()
if(TARGET Exiv2::exiv2lib)
  target_link_libraries(ikview_core PRIVATE Exiv2::exiv2lib)
  target_compile_definitions(ikview_core PRIVATE IKVIEW_HAS_EXIV2=1)
elseif(TARGET exiv2lib)
  target_link_libraries(ikview_core PRIVATE exiv2lib)
  target_compile_definitions(ikview_core PRIVATE IKVIEW_HAS_EXIV2=1)
endif()

add_library(ikview_ui STATIC src/imagebackground.h src/imagebackground.cpp src/viewerstatusbar.h src/viewerstatusbar.cpp src/viewerstyle.qrc src/viewerplaceholder.h src/viewerplaceholder.cpp src/dialoglayout.h src/dialoglayout.cpp src/viewericons.h src/viewericons.cpp src/viewertoolbutton.h src/viewertoolbutton.cpp src/imagecanvas.cpp src/imagecanvas.h src/mainwindow.cpp src/mainwindow.h src/apptheme.h src/apptheme.cpp src/preferencesdialog.h src/preferencesdialog.cpp src/shortcutmanager.h src/shortcutmanager.cpp src/exportdialog.h src/exportdialog.cpp src/finderhandoff.h src/finderhandoff.cpp)
if(APPLE)
  target_sources(ikview_ui PRIVATE src/systemintegration.mm src/macosservices.mm src/macosregistration.mm)
  target_link_libraries(ikview_ui PRIVATE "-framework AppKit" "-framework IOKit" "-framework CoreServices" "-framework UniformTypeIdentifiers")
else()
  target_sources(ikview_ui PRIVATE src/systemintegration.cpp)
endif()
target_include_directories(ikview_ui PUBLIC src)
target_link_libraries(ikview_ui PUBLIC ikview_core Qt6::Widgets Qt6::Quick Qt6::PrintSupport)
qt_add_executable(ikview MACOSX_BUNDLE WIN32 src/main.cpp)
target_link_libraries(ikview PRIVATE ikview_ui)
set_target_properties(ikview PROPERTIES
  MACOSX_BUNDLE_BUNDLE_NAME "ikview"
  MACOSX_BUNDLE_GUI_IDENTIFIER "io.ikview.desktop"
  MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
  MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
  MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/assets/Info.plist.in"
)
if(APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/AppIcon.icns")
  target_sources(ikview PRIVATE assets/AppIcon.icns)
  set_source_files_properties(assets/AppIcon.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(ikview PROPERTIES MACOSX_BUNDLE_ICON_FILE AppIcon.icns)
endif()
if(APPLE)
  target_sources(ikview PRIVATE assets/integration/install.sh assets/integration/ikview)
  set_source_files_properties(assets/integration/install.sh assets/integration/ikview PROPERTIES MACOSX_PACKAGE_LOCATION Resources/integration)
endif()
if(MSVC)
  target_compile_options(ikview_core PRIVATE /W4 /utf-8)
  target_compile_options(ikview_ui PRIVATE /W4 /utf-8)
  target_compile_options(ikview PRIVATE /utf-8)
else()
  target_compile_options(ikview_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(ikview_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT ikview_ipo)
if(ikview_ipo)
  set_property(TARGET ikview_core ikview_ui ikview PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/document_tests.cpp")
    add_executable(document_tests tests/document_tests.cpp)
    target_link_libraries(document_tests PRIVATE ikview_core Qt6::Test Qt6::Widgets)
    add_test(NAME document_tests COMMAND document_tests)
    set_tests_properties(document_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
  add_executable(canvas_tests tests/canvas_tests.cpp)
  target_link_libraries(canvas_tests PRIVATE ikview_ui Qt6::Test)
  add_test(NAME canvas_tests COMMAND canvas_tests)
  add_executable(ux_tests tests/ux_tests.cpp)
  target_link_libraries(ux_tests PRIVATE ikview_ui Qt6::Test)
  add_test(NAME ux_tests COMMAND ux_tests)
  add_executable(input_tests tests/input_tests.cpp)
  target_link_libraries(input_tests PRIVATE ikview_ui Qt6::Test)
  add_test(NAME input_tests COMMAND input_tests)
  set_tests_properties(canvas_tests ux_tests input_tests PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
  foreach(test_target IN ITEMS shortcut_tests preferences_tests preferences_integration_tests imagefilter_tests export_tests workflow_tests finder_handoff_tests)
    add_executable(${test_target} tests/${test_target}.cpp)
    target_link_libraries(${test_target} PRIVATE ikview_ui Qt6::Test)
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES RUN_SERIAL TRUE TIMEOUT 90)
  endforeach()
  set_tests_properties(canvas_tests ux_tests input_tests preferences_tests preferences_integration_tests imagefilter_tests export_tests workflow_tests
    PROPERTIES LABELS "desktop-gui")
  set_tests_properties(shortcut_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_executable(shutdown_tests tests/shutdown_tests.cpp)
  target_link_libraries(shutdown_tests PRIVATE ikview_ui Qt6::Test)
  add_test(NAME shutdown_tests COMMAND shutdown_tests)
  set_tests_properties(shutdown_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 60 LABELS "headless-ui")
  foreach(test_target IN ITEMS icon_tests toolbar_layout_tests design_tests)
    add_executable(${test_target} tests/${test_target}.cpp)
    target_link_libraries(${test_target} PRIVATE ikview_ui Qt6::Test)
    add_test(NAME ${test_target} COMMAND ${test_target})
    set_tests_properties(${test_target} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30 LABELS "headless-ui")
  endforeach()
  if(APPLE)
    add_executable(pdf_tests tests/pdf_tests.cpp)
    target_link_libraries(pdf_tests PRIVATE ikview_core Qt6::Test Qt6::Widgets "-framework CoreGraphics")
    add_test(NAME pdf_tests COMMAND pdf_tests)
    set_tests_properties(pdf_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
    add_executable(macosservices_tests tests/macosservices_tests.mm)
    target_link_libraries(macosservices_tests PRIVATE ikview_ui Qt6::Test "-framework AppKit")
    add_test(NAME macosservices_tests COMMAND macosservices_tests)
    set_tests_properties(macosservices_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(macosregistration_tests tests/macosregistration_tests.mm)
    target_link_libraries(macosregistration_tests PRIVATE ikview_ui Qt6::Test "-framework AppKit" "-framework UniformTypeIdentifiers")
    add_test(NAME macosregistration_tests COMMAND macosregistration_tests)
    set_tests_properties(macosregistration_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
      add_test(NAME integration_install_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration_install_tests.py")
      set_tests_properties(integration_install_tests PROPERTIES TIMEOUT 60)
      add_test(NAME license_tests COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/license_tests.py")
      set_tests_properties(license_tests PROPERTIES TIMEOUT 60)
    endif()
  endif()
  add_executable(ikview_benchmark tests/benchmark.cpp)
  target_link_libraries(ikview_benchmark PRIVATE ikview_core Qt6::Widgets)
  add_executable(ikview_responsiveness tests/responsiveness_benchmark.cpp)
  target_link_libraries(ikview_responsiveness PRIVATE ikview_ui Qt6::Test)
  add_executable(ikview_document_performance tests/document_performance.cpp)
  target_link_libraries(ikview_document_performance PRIVATE ikview_core Qt6::Widgets)
  add_executable(ikview_fixtures tests/format_fixtures.cpp)
  target_link_libraries(ikview_fixtures PRIVATE Qt6::Core Qt6::Gui LibArchive::LibArchive)
endif()
install(TARGETS ikview BUNDLE DESTINATION . RUNTIME DESTINATION bin)

include(cmake/finder-extension.cmake)
